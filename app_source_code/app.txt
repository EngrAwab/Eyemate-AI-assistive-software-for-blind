/* ───────────────────────── Main Activity ───────────────────────── */
class MainActivity : ComponentActivity() {

    /* ─────────────── Servers & text state ─────────────── */
    private var webServer: SimpleTextServer? = null
    private var serverText by mutableStateOf("")
    private var prevServerText = ""

    private var nanoViewServer: NanoViewServer? = null
    private var nanoViewReceivedMessage by mutableStateOf("")

    /* ─────────────── UI / pager state ─────────────── */
    private var currentPageIndex by mutableStateOf(0)
    private var lastPressTime = 0L

    /* ─────────────── TTS ─────────────── */
    @Volatile private var pcSpeechActive = false
    private var pcMessageCount = 0
    private lateinit var tts: TextToSpeech

    /* ─────────────── STT launcher ─────────────── */
    private lateinit var speechLauncher: ActivityResultLauncher<Intent>

    /* ─────────────── Phrases ─────────────── */
    private val computerPhrases = listOf(
        "", "Sleep", "Read", "Time", "Object Detection",
        "Guide Me", "Path Planning", "Live Guide", "Digital Eyes", "Mode"
    )
    private val piPhrases = listOf(
        "", "Sleep", "Read", "Time", "Object Detection",
        "Live Guide", "Digital Eyes"
    )
    private val levelPhrases = listOf(
        "Level 1", "Level 2", "Level 3", "Level 4",
        "Level 5", "Level 6", "Level 7", "Level 8"
    )
    private val level5Phrases = listOf(
        "stairs1", "stairs2", "stairs3", "stairs4", "student washroom",
        "Room 502 A", "Room 502 B", "Room 504 A", "Room 504 B", "Room 501 A",
        "Room 501 B", "Room 503 A", "Room 503 B", "faculty lift", "computer lab1",
        "computer lab2", "civil department", "faculty office", "media room", "FCR",
        "principal NUSET office", "female staff washroom", "male staff washroom",
        "department office1", "CS department", "principal NUSIT office",
        "faculty kitchen", "department office 2", "dean office", "CEN department",
        "EE department", "Conference room2"
    )

    /* ───────────────────────── onCreate ───────────────────────── */
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        /* last used mode */
        val prefs = getSharedPreferences("my_app", Context.MODE_PRIVATE)
        val initialMode = prefs.getString("last_mode", "computer") ?: "computer"

        /* ---------- TTS ---------- */
        tts = TextToSpeech(this) {
            if (it == TextToSpeech.SUCCESS) {
                tts.language = Locale.US
                tts.setSpeechRate(1.25f)
                tts.speak(
                    if (initialMode == "computer") "DeskMate" else "PieMate",
                    TextToSpeech.QUEUE_FLUSH, null, "init"
                )
            }
        }
        tts.setOnUtteranceProgressListener(object : UtteranceProgressListener() {
            override fun onStart(utteranceId: String?) {}
            override fun onDone(utteranceId: String?)  { finishPcUtterance(utteranceId) }
            override fun onError(utteranceId: String?) { finishPcUtterance(utteranceId) }
        })

        /* ---------- STT launcher ---------- */
        speechLauncher =
            registerForActivityResult(ActivityResultContracts.StartActivityForResult()) { result ->
                if (result.resultCode == RESULT_OK && result.data != null) {
                    val text = result.data
                        ?.getStringArrayListExtra(RecognizerIntent.EXTRA_RESULTS)
                        ?.firstOrNull()
                    if (!text.isNullOrBlank()) {
                        prevServerText = serverText
                        serverText     = "$text%"
                        webServer?.updateText(serverText)
                        Handler(Looper.getMainLooper()).postDelayed({
                            serverText = prevServerText
                            webServer?.updateText(prevServerText)
                        }, 500)
                    } else {
                        // Vibrate for 1 second if no speech recognized
                        val vibrator = getSystemService(VIBRATOR_SERVICE) as Vibrator
                        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                            vibrator.vibrate(VibrationEffect.createOneShot(1000, VibrationEffect.DEFAULT_AMPLITUDE))
                        } else {
                            @Suppress("DEPRECATION")
                            vibrator.vibrate(1000)
                        }
                    }
                }
            }

        /* ---------- HTTP servers ---------- */
        startServer()
        if (initialMode == "PiMate") startNanoViewServer()

        /* ---------- Compose UI ---------- */
        setContent {

            val navController = rememberNavController()
            val navBack by navController.currentBackStackEntryAsState()
            val currentRoute = navBack?.destination?.route

            fun speakWhenIdle(text: String, tag: String) {
                if (!pcSpeechActive && text.isNotBlank()) {
                    tts.stop()
                    tts.speak(text, TextToSpeech.QUEUE_FLUSH, null, tag)
                }
            }
            fun announce(mode: String, idx: Int) = speakWhenIdle(
                if (mode == "computer") computerPhrases.getOrElse(idx) { "" }
                else                     piPhrases.getOrElse(idx) { "" }, "pager"
            )
            fun announceLevel(idx: Int)  = speakWhenIdle(levelPhrases.getOrElse(idx){""},"level")
            fun announceLevel5(idx: Int) = speakWhenIdle(level5Phrases.getOrElse(idx){""},"level5")

            Scaffold(
                topBar = {
                    if (currentRoute in listOf("navigation", "level5"))
                        NavigationAppBar {
                            handleExitButtonPress(); navController.popBackStack()
                        }
                    else
                        AppBar(
                            onComputerSelected = {
                                prefs.edit().putString("last_mode", "computer").apply()
                                speakWhenIdle("DeskMate", "page")
                                navController.navigate("computer") {
                                    popUpTo(navController.graph.startDestinationId)
                                }
                                stopNanoViewServer(); startServer()
                            },
                            onNanoViewSelected = {
                                prefs.edit().putString("last_mode", "PiMate").apply()
                                handleStopButtonPress()
                                speakWhenIdle("PieMate", "page")
                                navController.navigate("PiMate") { launchSingleTop = true }
                                startNanoViewServer()
                            },
                            onInfoSelected = { navController.navigate("info") },
                            onStopPressed  = { handleStopButtonPress() }
                        )
                },
                floatingActionButton = { MicFab { startVoiceInput() } }
            ) { padding ->

                NavHost(navController, initialMode, Modifier.padding(padding)) {

                    /* ───────── DeskMate ───────── */
                    composable("computer") {
                        ComputerPage(
                            onLaunchDroidCam = { launchDroidCam() },
                            onStartServer    = { startServer() },
                            onButtonPress    = { handleButtonPress(it) },
                            onButtonRelease  = { updateDefaultText(currentPageIndex) },
                            onSleepPressed   = { handleSleepButtonPress() },
                            onStopPressed    = { handleStopButtonPress() },
                            onPageScroll = { idx ->
                                currentPageIndex = idx
                                updateDefaultText(idx)
                                vibratePhone()
                                announce("computer", idx)
                            },
                            onNavigate = { navController.navigate("navigation") }
                        )
                    }

                    /* ───────── PiMate ───────── */
                    composable("PiMate") {
                        NanoViewPage(
                            onLaunchDroidCam = { launchDroidCam() },
                            onStartServer    = { startNanoViewServer() },
                            onButtonPress    = { handleButtonPress(it) },
                            onButtonRelease  = { updateDefaultText(currentPageIndex) },
                            onSleepPressed   = { handleSleepButtonPress() },
                            onPageScroll = { idx ->
                                currentPageIndex = idx
                                updateDefaultText(idx)
                                vibratePhone()
                                announce("PiMate", idx)
                            },
                            receivedMessage  = nanoViewReceivedMessage
                        )
                    }

                    /* ───────── Level selector ───────── */
                    composable("navigation") {
                        NavigationPage(
                            onLevelSelected = {
                                handleButtonPress(it)
                                if (it == "lvl5") navController.navigate("level5")
                            },
                            onPageScroll = { idx ->
                                vibratePhone(); announceLevel(idx)
                            }
                        )
                    }

                    /* ───────── Level 5 ───────── */
                    composable("level5") {
                        Level5Page(
                            onNumberSelected = { handleButtonPress(it) },
                            onPageScroll     = { idx ->
                                vibratePhone(); announceLevel5(idx)
                            },
                            onExitPressed    = {
                                handleExitButtonPress(); navController.popBackStack()
                            }
                        )
                    }

                    /* ───────── Info ───────── */
                    composable("info") { InfoPage() }
                }
            }
        }
    }

    /* ──────────────────  STT helpers  ────────────────── */
    private fun ensureAudioPermission(): Boolean {
        return if (ActivityCompat.checkSelfPermission(
                this, Manifest.permission.RECORD_AUDIO
            ) == PackageManager.PERMISSION_GRANTED
        ) true
        else {
            ActivityCompat.requestPermissions(
                this, arrayOf(Manifest.permission.RECORD_AUDIO), 23
            )
            false
        }
    }

    private fun startVoiceInput() {
        if (!ensureAudioPermission()) return

        val intent = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply {
            putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL,
                RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
            putExtra(RecognizerIntent.EXTRA_LANGUAGE, Locale.getDefault())
            putExtra(RecognizerIntent.EXTRA_PROMPT, "Speak now…")
        }
        try {
            speechLauncher.launch(intent)
        } catch (e: Exception) {
            Toast.makeText(this, "Speech recognition not supported", Toast.LENGTH_SHORT).show()
        }
    }

    /* ────────────────── Core helpers ────────────────── */
    private fun finishPcUtterance(id: String?) {
        if (id?.startsWith("pc_") == true) {
            synchronized(this) {
                pcMessageCount--
                if (pcMessageCount <= 0) { pcMessageCount = 0; pcSpeechActive = false }
            }
        }
    }

    private fun handleStopButtonPress() {
        updateTextWithDelay("stop")
        tts.stop()
        tts.speak("", TextToSpeech.QUEUE_ADD, null, "empty")
    }

    private fun handleExitButtonPress() = updateTextWithDelay("lvlexit")

    private fun launchDroidCam() {
        val pkg = "com.dev47apps.droidcam"
        packageManager.getLaunchIntentForPackage(pkg)?.let(::startActivity)
            ?: Toast.makeText(this, "DroidCam app not found!", Toast.LENGTH_LONG).show()
    }

    private fun startServer() {
        if (webServer == null) thread {
            val ip = getIPAddress(); val port = 8080
            webServer = SimpleTextServer(port) { serverText }.apply { start() }
            runOnUiThread {
                Toast.makeText(
                    this, "DeskMate server: http://$ip:$port", Toast.LENGTH_LONG
                ).show()
            }
        }
    }

    private fun startNanoViewServer() {
        if (nanoViewServer == null) thread {
            val port = 8081
            nanoViewServer = NanoViewServer(port) { msg ->
                runOnUiThread {
                    nanoViewReceivedMessage = msg
                    val id = "pc_${System.currentTimeMillis()}"
                    synchronized(this) { pcMessageCount++; pcSpeechActive = true }
                    tts.speak(msg, TextToSpeech.QUEUE_ADD, null, id)
                }
            }.apply { start() }
            val ip = getIPAddress()
            runOnUiThread {
                Toast.makeText(
                    this, "PiMate server: http://$ip:$port", Toast.LENGTH_LONG
                ).show()
            }
        }
    }

    private fun stopNanoViewServer() { nanoViewServer?.stop(); nanoViewServer = null }

    private fun handleButtonPress(text: String) =
        if (text == "Mode") handleModeButtonPress() else updateTextWithDelay(text)

    private fun handleModeButtonPress() = updateTextWithDelay("Mode")

    private fun updateTextWithDelay(newText: String) {
        val now = System.currentTimeMillis()
        if (now - lastPressTime > 500) {
            serverText = newText
            webServer?.updateText(newText)
            Handler(Looper.getMainLooper()).postDelayed({
                updateDefaultText(currentPageIndex)
            }, 500)
        }
        lastPressTime = now
    }

    private fun updateDefaultText(idx: Int) {
        val mode = getSharedPreferences("my_app", Context.MODE_PRIVATE)
            .getString("last_mode", "computer") ?: "computer"

        val texts = if (mode == "computer") listOf(
            "", "sleep_n", "Read text_n", "Time_n",
            "Obstacle Detection_n", "nav_n", "path_n",
            "liveguide_n", "vlm_n", "Mode_n"
        ) else listOf(
            "", "sleep_n", "Read text_n", "Time_n",
            "Obstacle Detection_n", "liveguide_n", "vlm_n"
        )

        serverText = texts.getOrElse(idx) { "" }
        webServer?.updateText(serverText)
    }

    private fun vibratePhone() {
        ContextCompat.getSystemService(this, Vibrator::class.java)?.let { v ->
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O)
                v.vibrate(VibrationEffect.createOneShot(50, VibrationEffect.DEFAULT_AMPLITUDE))
            else @Suppress("DEPRECATION") v.vibrate(50)
        }
    }

    private fun getIPAddress(): String = try {
        NetworkInterface.getNetworkInterfaces().toList()
            .flatMap { it.inetAddresses.toList() }
            .firstOrNull { !it.isLoopbackAddress && !it.hostAddress.contains(":") }
            ?.hostAddress ?: "Unknown"
    } catch (e: Exception) { "Unknown" }

    private fun handleSleepButtonPress() {
        serverText = "Sleep"
        webServer?.updateText("Sleep")
        Handler(Looper.getMainLooper()).postDelayed({
            updateDefaultText(1)
        }, 500)
    }

    /* ────────────────── lifecycle ────────────────── */
    override fun onDestroy() {
        tts.stop(); tts.shutdown()
        super.onDestroy()
    }
}

/* ───────────────────────── NavigationPage ───────────────────────── */
@OptIn(ExperimentalPagerApi::class)
@Composable
fun NavigationPage(
    onLevelSelected: (String) -> Unit,
    onPageScroll: (Int) -> Unit
) {
    val pagerState = rememberPagerState()
    var dragging by remember { mutableStateOf(false) }

    Column(Modifier.fillMaxSize()) {
        HorizontalPager(
            count = 8,
            state = pagerState,
            modifier = Modifier
                .fillMaxWidth()
                .height(200.dp)
                .padding(3.dp)
                .pointerInput(Unit) {
                    detectHorizontalDragGestures(
                        onDragStart = { dragging = true },
                        onDragEnd = { dragging = false },
                        onHorizontalDrag = { _, _ -> }
                    )
                }
        ) { page ->
            val entries = listOf(
                "lvl1" to R.drawable.l1, "lvl2" to R.drawable.l2,
                "lvl3" to R.drawable.l3, "lvl4" to R.drawable.l4,
                "lvl5" to R.drawable.l5, "lvl6" to R.drawable.l6,
                "lvl7" to R.drawable.l7, "lvl8" to R.drawable.l8
            )
            val (name, res) = entries[page]
            var pressed by remember { mutableStateOf(false) }
            val y by animateDpAsState(if (pressed) 4.dp else 0.dp)

            Box(
                Modifier
                    .offset(y)
                    .fillMaxWidth()
                    .padding(4.dp)
                    .height(180.dp)
                    .border(2.dp, Color.White)
                    .pointerInput(Unit) {
                        detectTapGestures(
                            onPress = {
                                if (!dragging) {
                                    pressed = true
                                    try { awaitRelease() } finally { pressed = false }
                                } else awaitRelease()
                            },
                            onTap = {
                                if (!dragging) {
                                    ToneGenerator(AudioManager.STREAM_MUSIC, 100)
                                        .startTone(ToneGenerator.TONE_PROP_BEEP)
                                    onLevelSelected(name)
                                }
                            }
                        )
                    },
                contentAlignment = Alignment.Center
            ) {
                Image(
                    painterResource(id = res),
                    name,
                    Modifier.fillMaxSize(),
                    contentScale = ContentScale.Crop
                )
            }
        }
        LaunchedEffect(pagerState) {
            snapshotFlow { pagerState.currentPage }.collect { onPageScroll(it) }
        }
    }
}

/* ───────────────────────── HTTP server classes ───────────────────────── */
class SimpleTextServer(port: Int, private val textProvider: () -> String) : NanoHTTPD(port) {
    private var displayText: String = textProvider()
    override fun serve(session: IHTTPSession?): Response {
        if (session?.method == Method.OPTIONS) {
            val r = newFixedLengthResponse(Response.Status.OK, MIME_PLAINTEXT, "")
            addCors(r); return r
        }
        val resp = when (session?.method) {
            Method.GET, Method.POST -> newFixedLengthResponse(displayText)
            else -> newFixedLengthResponse(
                Response.Status.METHOD_NOT_ALLOWED,
                MIME_PLAINTEXT,
                "Method Not Allowed"
            )
        }
        addCors(resp); return resp
    }
    private fun addCors(r: Response) {
        r.addHeader("Access-Control-Allow-Origin", "*")
        r.addHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
        r.addHeader("Access-Control-Allow-Headers", "*")
    }
    fun updateText(txt: String) { displayText = txt }
}

class NanoViewServer(
    port: Int,
    private val onMessage: (String) -> Unit
) : NanoHTTPD(port) {
    override fun serve(session: IHTTPSession?): Response {
        if (session?.method == Method.OPTIONS) {
            val r = newFixedLengthResponse(Response.Status.OK, MIME_PLAINTEXT, "")
            addCors(r); return r
        }
        val resp = if (session?.method == Method.POST) {
            try { val files = mutableMapOf<String, String>(); session.parseBody(files) }
            catch (_: Exception) { }
            val msg = session?.parms?.get("message") ?: ""
            onMessage(msg)
            newFixedLengthResponse("received")
        } else newFixedLengthResponse("Send POST with 'message'")
        addCors(resp); return resp
    }
    private fun addCors(r: Response) {
        r.addHeader("Access-Control-Allow-Origin", "*")
        r.addHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
        r.addHeader("Access-Control-Allow-Headers", "*")
    }
}
